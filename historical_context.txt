# Theme Name: 霓虹科技

# Vibe & Description:
高饱和霓虹，深色基底，玻璃拟态光晕与彩色渐层叠加，营造现代、活力、前沿的数字科技感。

# Color

- 使用深紫蓝作为整体背景基底，推荐 `#1e1b4b` 或接近的深色，不使用纯白背景、灰色背景和黑色文字。
- 主色采用紫色 `#8b5cf6`，搭配青绿 `#06b6d4`、玫红 `#ec4899`、琥珀 `#f59e0b` 作为点缀。
- 大面积视觉重点使用渐变：`from-violet-500 via-purple-500 to-fuchsia-500`，用于按钮、标题强调、装饰线和关键数字。
- 卡片使用半透明玻璃效果：`bg-white/10`、`border-white/20`，文字使用 `text-white`、`text-white/80`、`text-white/60` 保持层次。

# Font

- 使用「阿里巴巴普惠体3.0」作为标题字体，突出科技感和信息层级。（https://resource-static.bj.bcebos.com/fonts-skill/AlibabaPuHuiTi_SemiBold.ttf）
- 使用「阿里巴巴普惠体3.0」作为正文字体，保持清晰、现代、易读。（https://resource-static.bj.bcebos.com/fonts-skill/AlibabaPuHuiTi_Regular.ttf）

- 元素出现时采用“淡入 + 光晕扩散”效果，渐变装饰可缓慢漂浮，营造动态科技氛围。

import React, { createContext, useContext, useState, useCallback, useRef } from 'react';
import type { Device, TraceItem, RunState } from '@/types/experiment';
import { traceItems as initialTraceItems } from '@/data/mockData';

interface ExperimentContextType {
  runId: string;
  runState: RunState;
  setRunState: (state: RunState) => void;
  sopStage: number;
  setSopStage: (stage: number) => void;
  sopCompleted: boolean;
  setSopCompleted: (v: boolean) => void;
  dataLoaded: boolean;
  setDataLoaded: (v: boolean) => void;
  visionStatus: 'PASS' | 'ANALYZING' | 'FAIL';
  setVisionStatus: (v: 'PASS' | 'ANALYZING' | 'FAIL') => void;
  visionConfidence: number;
  setVisionConfidence: (v: number) => void;
  devices: Device[];
  addDevice: (device: Device) => void;
  traceItems: TraceItem[];
  updateTraceItem: (index: number, detail: string, status: TraceItem['status']) => void;
  evidenceCount: number;
  setEvidenceCount: (v: number) => void;
  sample: { nuclei: number; rate: number; aggregation: number; perReaction: number };
  setSample: (s: { nuclei: number; rate: number; aggregation: number; perReaction: number }) => void;
  runSopSequence: () => void;
  sopTimerRef: React.MutableRefObject<number | null>;
}

const ExperimentContext = createContext<ExperimentContextType | null>(null);

export const ExperimentProvider: React.FC<{ children: React.ReactNode }> = ({ children }) => {
  const [runState, setRunState] = useState<RunState>('READY');
  const [sopStage, setSopStage] = useState(-1);
  const [sopCompleted, setSopCompleted] = useState(false);
  const [dataLoaded, setDataLoaded] = useState(false);
  const [visionStatus, setVisionStatus] = useState<'PASS' | 'ANALYZING' | 'FAIL'>('PASS');
  const [visionConfidence, setVisionConfidence] = useState(0.93);
  const [devices, setDevices] = useState<Device[]>([]);
  const [traceItems, setTraceItems] = useState<TraceItem[]>(initialTraceItems);
  const [evidenceCount, setEvidenceCount] = useState(8);
  const [sample, setSample] = useState({ nuclei: 182000, rate: 82, aggregation: 16, perReaction: 23500 });
  const sopTimerRef = useRef<number | null>(null);

  const addDevice = useCallback((device: Device) => {
    setDevices((prev) => [...prev, device]);
  }, []);

  const updateTraceItem = useCallback((index: number, detail: string, status: TraceItem['status']) => {
    setTraceItems((prev) => {
      const next = [...prev];
      if (next[index]) {
        next[index] = { ...next[index], detail, status };
      }
      return next;
    });
  }, []);

  const runSopSequence = useCallback(() => {
    if (sopTimerRef.current) {
      window.clearInterval(sopTimerRef.current);
      sopTimerRef.current = null;
    }
    setSopStage(-1);
    setSopCompleted(false);
    setSopStage(0);
    setRunState('RUNNING');
    let stage = 0;
    sopTimerRef.current = window.setInterval(() => {
      stage += 1;
      setSopStage(stage);
      setEvidenceCount(Math.min(8 + stage, 10));
      if (stage >= 6) {
        if (sopTimerRef.current) {
          window.clearInterval(sopTimerRef.current);
          sopTimerRef.current = null;
        }
        setSopCompleted(true);
        setSopStage(6);
        setRunState('COMPLETE');
        setTraceItems((prev) => {
          const next = [...prev];
          if (next[5]) next[5] = { ...next[5], detail: '数据产物已关联，等待 QC 结果', status: 'done' };
          return next;
        });
      }
    }, 620);
  }, []);

  return (
    <ExperimentContext.Provider
      value={{
        runId: 'SS-20260810-01',
        runState,
        setRunState,
        sopStage,
        setSopStage,
        sopCompleted,
        setSopCompleted,
        dataLoaded,
        setDataLoaded,
        visionStatus,
        setVisionStatus,
        visionConfidence,
        setVisionConfidence,
        devices,
        addDevice,
        traceItems,
        updateTraceItem,
        evidenceCount,
        setEvidenceCount,
        sample,
        setSample,
        runSopSequence,
        sopTimerRef,
      }}
    >
      {children}
    </ExperimentContext.Provider>
  );
};

export const useExperiment = () => {
  const ctx = useContext(ExperimentContext);
  if (!ctx) throw new Error('useExperiment must be used within ExperimentProvider');
  return ctx;
};
